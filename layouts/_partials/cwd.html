{{- /* Returns the shell working directory for a page, relative to ~ ("" for home). */ -}} {{- $cwd := "" -}} {{- if eq .Kind "section" -}} {{- $cwd = strings.TrimPrefix "/" .Path -}} {{- else if eq .Kind "page" -}} {{- with .Parent -}} {{- if not .IsHome -}}{{- $cwd = strings.TrimPrefix "/" .Path -}}{{- end -}} {{- end -}} {{- end -}} {{- return $cwd -}}
